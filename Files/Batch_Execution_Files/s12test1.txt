echo ---- Q1.2 AudiPeople ----
SELECT * FROM Events INTO AudiEvents WHERE location = Audi;
OPEN TABLE AudiEvents;
SELECT * FROM Participants JOIN AudiEvents INTO PartAudi WHERE Participants.event = AudiEvents.title;
CLOSE TABLE AudiEvents;
DROP TABLE AudiEvents;
OPEN TABLE PartAudi;
SELECT regNo, event FROM PartAudi INTO AudiPeople;
CLOSE TABLE PartAudi;
DROP TABLE PartAudi;
ALTER TABLE RENAME AudiPeople COLUMN event TO eventName;
echo ---- Q1.3 MiniEventPeople ----
SELECT * FROM Locations INTO MidLoc1 WHERE capacity >= 100;
OPEN TABLE MidLoc1;
SELECT * FROM MidLoc1 INTO MidLoc2 WHERE capacity <= 200;
CLOSE TABLE MidLoc1;

SELECT * FROM Events JOIN MidLoc2 INTO EventsMid WHERE Events.location = MidLoc2.name;
